function LetterKomAndTrim(BaseString, Symbol: string) : string;
var
	i 			: integer;
	KomRes		: string;
	NewString 	: string;
begin
	if (PostObj.DoingTime) then exit;
	KomRes := [' ',' '];
	i := Pos(Symbol, BaseString);
	if i = 0 then begin
		NewString := BaseString;
		BaseString := '';
	end else begin
		NewString := copy(BaseString, 1, i - 1);
		delete(BaseString, 1, i);
	end;
	NewString := Trim(NewString);
	KomRes[0] := BaseString;
	KomRes[1] := NewString;
	result := KomRes;
end;

function isValidDouble(ss: string): boolean;
var
	i	: integer;
begin
	if (PostObj.DoingTime) then exit;
	result := false;
	ss := trim(ss);
	if length(ss) > 0 then begin
		ss := StringReplace(ss,'0','',1); //1=rfReplaceAll 2=rfIgnoreCase
		ss := StringReplace(ss,'1','',1);
		ss := StringReplace(ss,'2','',1);
		ss := StringReplace(ss,'3','',1);
		ss := StringReplace(ss,'4','',1);
		ss := StringReplace(ss,'5','',1);
		ss := StringReplace(ss,'6','',1);
		ss := StringReplace(ss,'7','',1);
		ss := StringReplace(ss,'8','',1);
		ss := StringReplace(ss,'9','',1);

			exit;
		end;
	end else begin
		exit;
	end;

function ReadPSTLineVal(lineNum : string) : string;
var
    pstFile		: TStringlist;
    i			: integer;
	pos			: integer;
    pstLine		: string;
	variable	: string;
	value 		: string;
	num 		: string;
begin
	if (PostObj.DoingTime) then exit;
    Result := '';
	num := lineNum;
    { Load PST file }
    pstFile := TStringlist.Create;
    pstFile.LoadFromFile(PostObj.PSTfilename);
    { Parse through PST file }
    for i := 0 to pstFile.Count - 1 do begin
        pstLine := Trim(pstFile[i]);
        { Skip commented lines }
        if (AnsiPos(';', pstLine) = 1) then continue;
        { Find lines that contain values }
        pos := AnsiPos(':', pstLine);		
        if (pos > 0) then begin
            variable := Trim(Copy(pstLine, 1, pos - 1)); // everything left of the colon
            value := Copy(pstLine, pos + 1, Length(pstLine) - pos); // everything right of the colon
            { Find required line }			
            if (variable = num) or (variable = num + '.') or (variable = num + '.0') then begin
                pos := AnsiPos('=', value);
				
                { everything right of the equals }
                if (pos > 0) then value := Copy(value, pos + 1, Length(value) - pos);
                { Remove comments }
                pos := AnsiPos(';', value);
                if (pos > 0) then value := Copy(value, 1, pos - 1);
                { Return line value }
                Result := Trim(value);
                break;
            end;
        end;
    end;
    pstFile.Free;
end;

procedure ManageOVS;
var
	i         		: integer;
	PNCD      		: TNCData;
	PNCPunchPoint	: TNCPunchPoint;
begin
	if (PostObj.DoingTime) then exit;
	OVSX1 := 0;
	OVSY1 := 0;
	OVSX2 := 0;
	OVSY2 := 0;
	for i := 0 To PostObj.NCDataObjList.Count-1 do begin
		PNCD := PostObj.NCDataObjList.At_Object(i);
		if (PNCD.ProcessNum = 7) then begin
			if (ANSIUPPERCASE(PNCD.ToolName) = 'OVS') then begin
				if (ANSIUPPERCASE(PNCD.ClassName)='TNCPUNCHPOINT') then begin
					PNCPunchPoint := PNCD;
					if (OVSX1=0) and (OVSY1=0) then begin
						OVSX1 := PNCPunchPoint.xValAbs;
						OVSY1 := PNCPunchPoint.yValAbs;
					end else begin
						if (PNCPunchPoint.xValAbs=OVSX1) and (PNCPunchPoint.yValAbs=OVSY1) then begin
							//non fa nulla. a volte con la punzonatura singola SigmaNEST crea due colpi uno esattamente sopra all'altro,
							//questa if serve a evitare di registrare due OVS con identiche coordinate
						end else begin
							OVSX2 := PNCPunchPoint.xValAbs;
							OVSY2 := PNCPunchPoint.yValAbs;
						end;
					end;
				end;
				PNCD.Ignore := true;
			end;
		end;
	end;
	if (PostObj.PromptOpt2=4) then begin
		if (OVSX1=0) then OVSX1 := StrToFloat(InputBox('SigmaNEST','Primo colpo utensile OVS non trovato. Quota X1?','100'));
		if (OVSY1=0) then OVSY1 := StrToFloat(InputBox('SigmaNEST','Primo colpo utensile OVS non trovato. Quota Y1?','100'));
		if (OVSX2=0) then OVSX2 := StrToFloat(InputBox('SigmaNEST','Secondo colpo utensile OVS non trovato. Quota X2?','100'));
		if (OVSY2=0) then OVSY2 := StrToFloat(InputBox('SigmaNEST','Secondo colpo utensile OVS non trovato. Quota Y2?','100'));
	end;
end;

procedure EdgeDetectionMacro;
const
    ClampTolerance = 5;
var
	i	: string;
	j	: string;
	q	: string;
	a	: double;
	b	: double;
	c	: double;  
begin
	if ((PostObj.DoingTime) or (PostObj.SinglePartProgram)) then exit;
	case PostObj.PromptOpt2 of
		1,2,3,4,5 : begin
			a := (PostObj.SheetLength * 0.15);
			b := (PostObj.SheetLength * 0.8);
			c := (PostObj.SheetWidth * 0.7);
		end;
		6 : begin
			a := (PostObj.SheetWidth * 0.15);
			b := (PostObj.SheetLength * 0.5);
			c := (PostObj.SheetLength * 0.8);
		end;
		7 : begin
			a := (PostObj.SheetWidth * 0.15);
			b := (PostObj.SheetWidth * 0.7);
			c := (PostObj.SheetLength * 0.5);
		end;
	end;
	
	if (isValidInteger(PostObj.GetVariable('#251'))) then begin
        case PostObj.PromptOpt2 of
            1,2,3,4,5 : if (StrToInt(PostObj.GetVariable('#251'))<=PostObj.SheetLength) and (StrToInt(PostObj.GetVariable('#251'))>=0) then a := StrToInt(PostObj.GetVariable('#251'));
            6,7 :     if (StrToInt(PostObj.GetVariable('#251'))<=PostObj.SheetWidth) and (StrToInt(PostObj.GetVariable('#251'))>=0) then a := StrToInt(PostObj.GetVariable('#251'));
        end;
    end;
    
	if (isValidInteger(PostObj.GetVariable('#252'))) then begin
        case PostObj.PromptOpt2 of
            1,2,3,4,5,6 : if (StrToInt(PostObj.GetVariable('#252'))<=PostObj.SheetLength) and (StrToInt(PostObj.GetVariable('#252'))>=0) then b := StrToInt(PostObj.GetVariable('#252'));
            7 :         if (StrToInt(PostObj.GetVariable('#252'))<=PostObj.SheetWidth) and (StrToInt(PostObj.GetVariable('#252'))>=0) then b := StrToInt(PostObj.GetVariable('#252'));
        end;
    end;
	if (isValidInteger(PostObj.GetVariable('#253'))) then begin
        case PostObj.PromptOpt2 of
            1,2,3,4,5 : if (StrToInt(PostObj.GetVariable('#253'))<=PostObj.SheetWidth) and (StrToInt(PostObj.GetVariable('#253'))>=0) then c := StrToInt(PostObj.GetVariable('#253'));	
            6,7 :     if (StrToInt(PostObj.GetVariable('#253'))<=PostObj.SheetLength) and (StrToInt(PostObj.GetVariable('#253'))>=0) then c := StrToInt(PostObj.GetVariable('#253'));	
        end;
    end;
	{ClampX1                       	= 175;
	ClampX2                       	= 698;
	ClampX3                       	= 1750;
	ClampX4                       	= 2275;
	ClampY1                       	= 5000;}
  
    
 
	if (PostObj.PromptOpt2=2) or (PostObj.PromptOpt2=3) or (PostObj.PromptOpt2=4) then begin
		if (a > (ClampX1-(ClampSize/2))) and (a < (ClampX1+(ClampSize/2))) then a := ClampX1+(ClampSize/2)-ClampTolerance;
		if (a > (ClampX2-(ClampSize/2))) and (a < (ClampX2+(ClampSize/2))) then a := ClampX2+(ClampSize/2)-ClampTolerance;	
		if (a > (ClampX3-(ClampSize/2))) and (a < (ClampX3+(ClampSize/2))) then a := ClampX3-(ClampSize/2)-ClampTolerance;	
		if (a > (ClampX4-(ClampSize/2))) and (a < (ClampX4+(ClampSize/2))) then a := ClampX4-(ClampSize/2)-ClampTolerance;	
        
		if (b > (ClampX1-(ClampSize/2))) and (b < (ClampX1+(ClampSize/2))) then b := ClampX1+(ClampSize/2)-ClampTolerance;
		if (b > (ClampX2-(ClampSize/2))) and (b < (ClampX2+(ClampSize/2))) then b := ClampX2+(ClampSize/2)-ClampTolerance;	
		if (b > (ClampX3-(ClampSize/2))) and (b < (ClampX3+(ClampSize/2))) then b := ClampX3-(ClampSize/2)-ClampTolerance;	
		if (b > (ClampX4-(ClampSize/2))) and (b < (ClampX4+(ClampSize/2))) then b := ClampX4-(ClampSize/2)-ClampTolerance;
        
		if (c > (ClampY1-(ClampSize/2))) and (c < (ClampY1+(ClampSize/2))) then c := ClampY1-(ClampSize/2)-ClampTolerance;		
	end;
	
	i := 'I'+FormatFloat('00.0_',a);
	j := 'J'+FormatFloat('00.0_',b);
	q := 'Q'+FormatFloat('00.0_',c);
	
	case PostObj.PromptOpt2 of
		1 : begin
			//do nothing
		end;
		2 : begin //g121

procedure Add_E_Value;															// E management
var
	var102			: string;
	NewX			: real;
	NewY			: real;
	AngoloRad		: real;
	AttaccoCorto	: real;
begin
	if (PostObj.DoingTime) then exit;
	if ((PostObj.Vaporizing) or (PostObj.ToolNumber = 5)) then exit;
	if PostObj.toolnumber < 16  then begin	   // To include also etching and marking
		var102 := PostObj.GetVariable('#102');

			1 : var102 := '1';
			2 : var102 := '2';
			3 : var102 := '3';
			6 : var102 := '4';
			9 : var102 := '5';
			12 : var102 := '6';
			13 : var102 := '7';
			14 : var102 := '8';
			15 : var102 := '9';
                     
		end;

			// Fly Piercing
			if (PostObj.PromptOpt6=2) and (FlyPiercing_M105 = 1) then begin				// Flying pierce ALWAYS ON by prompt option

				end;

			if ((PostObj.PromptOpt6=1) and (FlyPiercing_M105 = 1) and (AttaccoCorto>0)) then begin		    	// Flying pierce auto management by prompt option
				if (PostObj.preg_Code=0) and (PostObj.EntType=1) and (SfondaVeloce = True) then begin			// on leadin and only on head down fast pierce
					if (PostObj.EntityLength > (AttaccoCorto+FlyingPiercingMinLength)) then begin					// if leadin is long enough to be broken
						if AttaccoCorto = 0 then AttaccoCorto := 0.75;					// default just in case
						AngoloRad := DegtoRad(PostObj.LineAngle); //(pi / 180 * PostObj.LineAngle);
						NewX := PostObj.PreX + (AttaccoCorto * cos(AngoloRad));
						NewY := PostObj.PreY + (AttaccoCorto * sin(AngoloRad));
						PostObj.AddToNCFile('E' + inttostr(FlyingPiercingENum) + '|G01X' + formatfloat('0.00',NewX) + 'Y' + formatfloat('0.00',NewY));  			   
					end else begin																// leadin too short -> ALL in E8
						var102 := FlyingPiercingENum;
					end;	
				end;
			end;	  
		end;	
		if (PostObj.ToolNumber=5) then var102 := '9';										// special case forced for customer ???
		if (var102 <> '') then begin
			currentmodalE := StrToInt(var102);
		end;
		if ((OutputFlashCutVar) or ((UseFlashCutEForFlyCut) and (PostObj.EntSubAttr = 27))) then begin
			var102 := FlashCutE;
		end;
		if (premodalE <> currentmodalE) then begin
			PostObj.Var8 := 'E' + var102;
			premodalE := currentmodalE;
		end else begin
			PostObj.Var8 := '';
		end;
	end else begin
		premodalE := 0;
	end;
end;

procedure FlameOff;
var
	ZPos 		: string;
	Speciale	: integer;
	M01			: string;
begin
	if (PostObj.DoingTime) then exit;
	ZPos := '';
	Speciale := 0;
	M01 := '';
	// Manage head-up when changing process
	if (GetPreEntityProcessNum <> PostObj.ToolNumber) then begin
		PostObj.HeadDown:=false;
		SfondaVeloce := False;
	end;
	if (EndOfSub) then begin
		PostObj.HeadDown := false;
		if EnableM01 then M01 := '_M01';
		EndOfSub := false;
	end;
	if (PartCompleted) then begin
		PostObj.HeadDown := false;
		if (PostObj.WritingSub<>2) then begin
			if EnableM01 then M01 := '_M01';
		end;
		PartCompleted := false;
	end;
	if (PostObj.ToolNumber > 4) and (PostObj.ToolNumber < 10) then begin
		if (PostObj.SideAttr = 0) then begin
			Speciale := 1;
		end;
	end;
	if (PostObj.EntSubAttr=27) and (PostObj.EntLength<30) then begin //atSub_FlyCut
		PostObj.HeadDown:=true;
	end;
	if (GetPreEntityProcessNum=10) and (PostObj.ToolNumber=10) and (PostObj.EntLength<30) then begin
		PostObj.HeadDown := true;
		SfondaVeloce := True;
	end else begin
		if (PostObj.PromptOpt3=1) then begin
			PostObj.HeadDown := False;
			Speciale := 0;
		end;
		if (PostObj.HeadDown = True) and (Speciale = 0) then begin                // se il prossimo movimento � in testa bassa
			case PostObj.PromptOpt1 of
				'1','4': begin
					SfondaVeloce := True;
				end;                                                 
				'2': begin	
					if (PostObj.EntSubAttr=27) then begin //atSub_FlyCut
						SfondaVeloce := True;
					end else begin
						if not (ThereWasAStopBeforeFlameOff) then begin
							if EnableM01 then begin
								PostObj.AddToNCFile(FlameOff_Code+M01);                 //Beam OFF Head down
							end else begin
								PostObj.AddToNCFile(FlameOff_Code);
							end;	
						end;
						SfondaVeloce := False;
					end;
				end;
				'3':begin			
					if (PostObj.EntSubAttr=27) then begin //atSub_FlyCut
						SfondaVeloce := True;
					end else begin
						ZPos := 'Z'+PostObj.GetVariable('#105');   //Beam OFF Head down
						PostObj.AddToNCFile(FlameOff_Code+ZPos);         			
						SfondaVeloce := False;                     			
					end;						
				end;
			end;   // end case		
		end else begin                                                 // testa alta
			if not (ThereWasAStopBeforeFlameOff) then begin
				if EnableM01 then begin
					PostObj.AddToNCFile(FlameOff_Code+M01); //+ZPos); //beam off head up
				end else begin
					PostObj.AddToNCFile(FlameOff_Code);
				end;
			end;
			SfondaVeloce := False;
		end;
	end;
	ThereWasAStopBeforeFlameOff := false;
end;

function AddZeroToLeft(aNumber : double; ResLength : integer) : string;
var
	i 		: integer;
	strNum 	: string;
	resVal	: string;
begin
	strNum := FormatFloat('0.00',aNumber);
	resVal := strNum;
	for i := length(strNum) to ResLength - 1 do begin
		resVal := '0'+resVal;
	end;
	result := resVal;
end;

procedure ProcessFlashCut(FlashStart,FlashEnd : integer);
var
	MinX,MinY,MaxX,MaxY		: double;
	XDim, YDim				: double;
	FlameOffXDim			: double;
	FlameOffYDim			: double;
	PNCD					: TNCData;
    i,j                     : integer;
	IndexList				: TStringList;
	XDimList				: TStringList;
	YDimList				: TStringList;
	XCoorList				: TStringList;
	YCoorList				: TStringList;
	tmpFlameOffXDim			: double;
	tmpFlameOffYDim			: double;
	pNCLine					: TNCLine;
begin
	if PostObj.doingTime then exit;
	XDim:=0;
	YDim:=0;
	IndexList := TStringList.Create;
	XDimList := TStringList.Create;
	YDimList := TStringList.Create;
	XCoorList := TStringList.Create;
	YCoorList := TStringList.Create;
	XCoorList.Sorted := true;
	YCoorList.Sorted := true;
	XCoorList.Duplicates := dupIgnore;
	YCoorList.Duplicates := dupIgnore;
	for i := FlashStart to FlashEnd-1 do begin
		PNCD := PostObj.NCDataObjList.At_Object(i);
        IndexList.Add(IntToStr(i));
        if (PNCD.GCode = 1) and (PNCD.ClassName='TNCLine') then begin
            if (PNCD.CutterComp=0) then begin
                IndexList.Clear;
                XDimList.Clear;
                YDimList.Clear;
                XCoorList.Clear;
                YCoorList.Clear;
                FlashCutList.Clear;
                exit;
            end;
            pNCLine := PNCD;
            if (abs(pNCLine.xStart - pNCLine.xValAbs)>0.01) and (abs(pNCLine.yStart - pNCLine.yValAbs)<0.01) then begin
                if (XDim = 0) then begin
                    XCoorList.Add(AddZeroToLeft(pNCLine.xStart,10));
                    YCoorList.Add(AddZeroToLeft(pNCLine.yStart,10));
                end;
                XDim := XDim + sqrt(((pNCLine.xValAbs-pNCLine.xStart)*(pNCLine.xValAbs-pNCLine.xStart))+((pNCLine.yValAbs-pNCLine.yStart)*(pNCLine.yValAbs-pNCLine.yStart)));
                if (YDim <> 0) then begin 
                    XCoorList.Add(AddZeroToLeft(pNCLine.xStart,10));
                    YCoorList.Add(AddZeroToLeft(pNCLine.yStart,10));
                    YDimList.Add(FormatFloat('0.00',YDim));
                    YDim := 0;
                end;
            end;
            if (abs(pNCLine.xStart - pNCLine.xValAbs)<0.01) and (abs(pNCLine.yStart - pNCLine.yValAbs)>0.01) then begin
                if YDim = 0 then begin
                    XCoorList.Add(AddZeroToLeft(pNCLine.xStart,10));
                    YCoorList.Add(AddZeroToLeft(pNCLine.yStart,10));
                end;
                YDim := YDim + sqrt(((pNCLine.xValAbs-pNCLine.xStart)*(pNCLine.xValAbs-pNCLine.xStart))+((pNCLine.yValAbs-pNCLine.yStart)*(pNCLine.yValAbs-pNCLine.yStart)));
                if XDim <> 0 then begin 
                    XCoorList.Add(AddZeroToLeft(pNCLine.xStart,10));
                    YCoorList.Add(AddZeroToLeft(pNCLine.yStart,10));
                    XDimList.Add(FormatFloat('0.00',XDim));
                    XDim := 0;
                end;
            end;
            if (abs(pNCLine.xStart - pNCLine.xValAbs)>0.01) and (abs(pNCLine.yStart - pNCLine.yValAbs)>0.01) then begin
                exit;
            end;
        end;
        if (PNCD.GCode = 0) then begin
            if (PNCD.ClassName='TNCLine') then begin
                if (XDim <> 0) and (YDim = 0) then begin 
                    PNCD := PostObj.NCDataObjList.At_Object(i-1);
                    pNCLine := PNCD;
                    XCoorList.Add(AddZeroToLeft(pNCLine.xValAbs,10));
                    YCoorList.Add(AddZeroToLeft(pNCLine.yValAbs,10));
                    XDimList.Add(FormatFloat('0.00',XDim));					
                end;
                if (YDim <> 0) and (XDim = 0) then begin
                    PNCD := PostObj.NCDataObjList.At_Object(i-1);
                    pNCLine := PNCD;
                    XCoorList.Add(AddZeroToLeft(pNCLine.xValAbs,10));
                    YCoorList.Add(AddZeroToLeft(pNCLine.yValAbs,10));
                    YDimList.Add(FormatFloat('0.00',YDim));					
                end;
                XDim := 0;
                YDim := 0;
            end;
        end;
	end;	
    //Inc(i);
    if (XDim <> 0) and (YDim = 0) then begin 
        PNCD := PostObj.NCDataObjList.At_Object(i-1);
        if (PNCD.ClassName='TNCLine') then begin
            pNCLine := PNCD;
            XCoorList.Add(AddZeroToLeft(pNCLine.xValAbs,10));
            YCoorList.Add(AddZeroToLeft(pNCLine.yValAbs,10));
            XDimList.Add(FormatFloat('0.00',XDim));
        end;
    end;
    if (YDim <> 0) and (XDim = 0) then begin
        PNCD := PostObj.NCDataObjList.At_Object(i-1);
        if (PNCD.ClassName='TNCLine') then begin
            pNCLine := PNCD;
            XCoorList.Add(AddZeroToLeft(pNCLine.xValAbs,10));
            YCoorList.Add(AddZeroToLeft(pNCLine.yValAbs,10));
            YDimList.Add(FormatFloat('0.00',YDim));
        end;				
    end;
    XDim := 0;
    YDim := 0;
    if IndexList.Count > 0 then begin
        MinX := 9999999;
        MinY := 9999999;
        MaxX := 0;
        MaxY := 0;
        //ciclo per trovare X e Y min e max della griglia. 
        for j := 1 to IndexList.Count - 1 do begin
            PNCD := PostObj.NCDataObjList.At_Object(StrToInt(IndexList[j]));
            if (PNCD.ClassName = 'TNCLine') then begin
                pNCLine := PNCD;
                if MinX > pNCLine.xStart then MinX := pNCLine.xStart; 
                if MinY > pNCLine.yStart then MinY := pNCLine.yStart; 
                if MaxX < pNCLine.xValAbs then MaxX := pNCLine.xValAbs;
                if MaxY < pNCLine.yValAbs then MaxY := pNCLine.yValAbs;
            end else begin
                IndexList.Clear;
                XDimList.Clear;
                YDimList.Clear;
                XCoorList.Clear;
                YCoorList.Clear;
                break;
            end;
        end; 
        //Controllo che tutti gli enti orizzontali siano di dimensione uguale
        for j := 1 to XDimList.Count -1 do begin
            if XDimList[j] <> XDimList[j-1] then begin
                IndexList.Clear;
                XDimList.Clear;
                YDimList.Clear;
                XCoorList.Clear;
                YCoorList.Clear;
                break;
            end;
        end;
        //Controllo che tutti gli enti verticali siano di dimensione uguale
        for j := 1 to YDimList.Count -1 do begin
            if YDimList[j] <> YDimList[j-1] then begin
                IndexList.Clear;
                XDimList.Clear;
                YDimList.Clear;
                XCoorList.Clear;
                YCoorList.Clear;
                break;
            end;
        end;
        if (((XCoorList.Count/2)*YCoorList.Count)<>XDimList.Count) then begin
            IndexList.Clear;
            XDimList.Clear;
            YDimList.Clear;
            XCoorList.Clear;
            YCoorList.Clear;
        end;
        if (((YCoorList.Count/2)*XCoorList.Count)<>YDimList.Count) then begin
            IndexList.Clear;
            XDimList.Clear;
            YDimList.Clear;
            XCoorList.Clear;
            YCoorList.Clear;
        end;
        FlameOffXDim:=0;
        for j := 1 to XCoorList.Count - 1 do begin
            if (Abs(StrToFloat(FormatFloat('0.00',Abs((StrToFloat(Trim(XCoorList[j])))-(StrToFloat(Trim(XCoorList[j-1]))))))-StrToFloat(XDimList[0]))>0.01) then begin
                if FlameOffXDim=0 then begin
                    FlameOffXDim := StrToFloat(FormatFloat('0.00',Abs((StrToFloat(Trim(XCoorList[j])))-(StrToFloat(Trim(XCoorList[j-1]))))));
                end else begin
                    if (Abs(StrToFloat(FormatFloat('0.00',Abs((StrToFloat(Trim(XCoorList[j])))-(StrToFloat(Trim(XCoorList[j-1]))))))-FlameOffXDim)>0.01) then begin
                        IndexList.Clear;
                        XDimList.Clear;
                        YDimList.Clear;
                        XCoorList.Clear;
                        YCoorList.Clear;
                        break;
                    end;
                end;
            end;
        end;				
        FlameOffYDim:=0;
        for j := 1 to YCoorList.Count - 1 do begin
            if (Abs(StrToFloat(FormatFloat('0.00',Abs((StrToFloat(Trim(YCoorList[j])))-(StrToFloat(Trim(YCoorList[j-1]))))))-StrToFloat(YDimList[0]))>0.01) then begin
                if FlameOffYDim=0 then begin
                    FlameOffYDim := StrToFloat(FormatFloat('0.00',Abs((StrToFloat(Trim(YCoorList[j])))-(StrToFloat(Trim(YCoorList[j-1]))))));
                end else begin
                    if (Abs(StrToFloat(FormatFloat('0.00',Abs((StrToFloat(Trim(YCoorList[j])))-(StrToFloat(Trim(YCoorList[j-1]))))))-FlameOffYDim)>0.01) then begin
                        IndexList.Clear;
                        XDimList.Clear;
                        YDimList.Clear;
                        XCoorList.Clear;
                        YCoorList.Clear;
                        break;
                    end;
                end;
            end;
        end;				
        if IndexList.Count > 0 then begin
            tmpFlameOffXDim := 0;
            tmpFlameOffYDim := 0;
            tmpFlameOffXDim	:= MaxX-MinX;
            tmpFlameOffXDim := (tmpFlameOffXDim-(StrToFloat(XDimList[0])*(XCoorList.Count/2)))/((XCoorList.Count/2)-1);
            tmpFlameOffYDim	:= MaxY-MinY;
            tmpFlameOffYDim := (tmpFlameOffYDim-(StrToFloat(YDimList[0])*(YCoorList.Count/2)))/((YCoorList.Count/2)-1);	
            if (FlameOffXDim=0) and (FormatFloat('0.00',tmpFlameOffXDim)=FormatFloat('0.00',StrToFloat(XDimList[0]))) then FlameOffXDim := tmpFlameOffXDim;
            if (FlameOffYDim=0) and (FormatFloat('0.00',tmpFlameOffYDim)=FormatFloat('0.00',StrToFloat(YDimList[0]))) then FlameOffYDim := tmpFlameOffYDim;
            if StrToFloat(XDimList[0])>=StrToFloat(YDimList[0]) then begin	
                FlashCutList.Add(IndexList[0]+';'+
                                 'G111 I'+XDimList[0]+' J'+YDimList[0]+';'+
                                 'G136 X'+FormatFloat('0.00',MinX+(StrToFloat(XDimList[0])/2))+
                                 ' Y'+FormatFloat('0.00',MinY+(StrToFloat(YDimList[0])/2))+
                                 ' I'+FormatFloat('0.00',(FlameOffXDim+StrToFloat(XDimList[0])))+
                                 ' J'+FormatFloat('0.00',(FlameOffYDim+StrToFloat(YDimList[0])))+
                                 ' P'+IntToStr((XCoorList.Count/2)-1)+
                                 ' K'+IntToStr((YCoorList.Count/2)-1)+
                                 ' Q0. L1;M104');
            end else begin
                FlashCutList.Add(IndexList[0]+';'+
                                 'G111 I'+YDimList[0]+' J'+XDimList[0]+' K90.'+';'+
                                 'G136 X'+FormatFloat('0.00',MaxX-(StrToFloat(XDimList[0])/2))+
                                 ' Y'+FormatFloat('0.00',MinY+(StrToFloat(YDimList[0])/2))+
                                 ' I'+FormatFloat('0.00',(FlameOffYDim+StrToFloat(YDimList[0])))+
                                 ' J'+FormatFloat('0.00',(FlameOffXDim+StrToFloat(XDimList[0])))+
                                 ' P'+IntToStr((YCoorList.Count/2)-1)+
                                 ' K'+IntToStr((XCoorList.Count/2)-1)+
                                 ' Q90. L1;M104');
            end;
            //Disabilito tutti gli oggetti del fly cut tranne il primo
            PNCD := PostObj.NCDataObjList.At_Object(StrToInt(IndexList[0]));
            GetNewLookupID(PNCD,5,0,0,0,0);
            for j := 1 to IndexList.Count - 1 do begin
                PNCD := PostObj.NCDataObjList.At_Object(StrToInt(IndexList[j]));
                PNCD.Ignore := true;
            end;
        end;
        IndexList.Clear;
        XDimList.Clear;
        YDimList.Clear;
        XCoorList.Clear;
        YCoorList.Clear;
    end;   
	IndexList.Free;
	XDimList.Free;
	YDimList.Free;
	XCoorList.Free;
	YCoorList.Free;
end;

procedure RecognizeFlashCut;
const
    DebugFC                                 = false;
var
	GridMinX,GridMinY,GridMaxX,GridMaxY		: double;
    NumOfSteps                              : integer;
    //step 0 happens when the flash cut path starts
    //step 1 happens when the path goes from horizontal to vertical (or vice versa) without turning off the laser
    //step 2 happens when the path gets back to its original starting point
    //step 3 happens when the path reaches the farthest x and y position from the starting point, which have been recognized and stored before step 1 took place
	i,j,IndexToSkip			                : integer;
	PNCD,PrePNCD,NextPNCD                   : TNCData;
	NCLine, PreNCLine						: TNCLine;
	PrePNCDAng, PNCDAng						: double;
begin
	if (PostObj.doingTime) then exit;
    if (DebugFC) then PostObj.AddToNCFile('procedure started');
    IndexToSkip := 0;
	for i := 1 to PostObj.NCDataObjList.Count - 2 do begin
        if (i<=IndexToSkip) then continue; //used to skip to the end of a previously recognized flash cut group
		PrePNCD := PostObj.NCDataObjList.At_Object(i-1);
		PNCD := PostObj.NCDataObjList.At_Object(i);
        NextPNCD := PostObj.NCDataObjList.At_Object(i+1);
        if (PNCD.ClassName = 'TNCLine') and (PNCD.EntSubAttr=27) and (PrePNCD.EntSubAttr<>27) then begin            
			NCLine := PNCD;
			NumOfSteps := 0;
            GridMinX := NCLine.xStart;
            GridMinY := NCLine.yStart;
            GridMaxX := NCLine.xStart;
            GridMaxY := NCLine.yStart;
            if (DebugFC) then PostObj.AddToNCFile('------------------flash cut start found at i='+inttostr(i)+#13#10+
                                                  'GridMinX='+floattostr(GridMinX)+#13#10+
                                                  'GridMinY='+floattostr(GridMinY)+#13#10+
                                                  'GridMaxX='+floattostr(GridMaxX)+#13#10+
                                                  'GridMaxY='+floattostr(GridMaxY)+#13#10+
                                                  'NumOfSteps='+inttostr(NumOfSteps)+#13#10+
												  'GCode='+IntToStr(NCLine.GCode));
            for j := i+1 to PostObj.NCDataObjList.Count - 1 do begin
                PrePNCD := PostObj.NCDataObjList.At_Object(j-1);
                PNCD := PostObj.NCDataObjList.At_Object(j);
				if (PNCD.ClassName = 'TNCLine') then begin
					PreNCLine := PrePNCD;
					NCLine := PNCD;
					if (PNCD.EntSubAttr=27) then begin
						PrePNCDAng := GetCoeffAng(PreNCLine.xStart, PreNCLine.yStart, PreNCLine.xValAbs, PreNCLine.yValAbs, 1);
						PNCDAng := GetCoeffAng(NCLine.xStart, NCLine.yStart, NCLine.xValAbs, NCLine.yValAbs, 1);
						if (DebugFC) then PostObj.AddToNCFile('flash cut entity at j='+inttostr(j)+#13#10+
															'PrePNCDAng='+floattostr(PrePNCDAng)+#13#10+
															'PNCDAng='+floattostr(PNCDAng)+#13#10+
															'PrePNCD.GCode='+IntToStr(PrePNCD.GCode)+#13#10+
															'PNCD.GCode='+IntToStr(PNCD.GCode)+#13#10);                       
						if (NCLine.xStart<GridMinX) then GridMinX := NCLine.xStart;
						if (NCLine.yStart<GridMinY) then GridMinY := NCLine.yStart;
						if (NCLine.xStart>GridMaxX) then GridMaxX := NCLine.xStart;
						if (NCLine.yStart>GridMaxY) then GridMaxY := NCLine.yStart;
						if (NCLine.xValAbs<GridMinX) then GridMinX := NCLine.xValAbs;
						if (NCLine.yValAbs<GridMinY) then GridMinY := NCLine.yValAbs;
						if (NCLine.xValAbs>GridMaxX) then GridMaxX := NCLine.xValAbs;
						if (NCLine.yValAbs>GridMaxY) then GridMaxY := NCLine.yValAbs;
						
						if (PrePNCD.EntSubAttr=27) and (PNCD.EntSubAttr=27) and (PrePNCD.GCode=1) and (PNCD.GCode=1) and (PNCDAng<>PrePNCDAng) and (NumOfSteps=0) then begin
							if (DebugFC) then PostObj.AddToNCFile('-----moving to STEP 1 at j='+inttostr(j));
							//step 1 happens when the path goes from horizontal to vertical (or vice versa) without turning off the laser
							NumOfSteps := 1; 
						end;
					end else begin
						if (DebugFC) then PostObj.AddToNCFile('regular entity at j='+inttostr(j));
						if (NumOfSteps>0) and                                           //alread moved for horizontal to vertical (or vice versa)
						   ((NCLine.xStart=GridMinX) and (NCLine.yStart=GridMinY) or       //reached the min vertex
						    (NCLine.xStart=GridMinX) and (NCLine.yStart=GridMaxY) or
						    (NCLine.xStart=GridMaxX) and (NCLine.yStart=GridMinY) or       //reached the min vertex
						    (NCLine.xStart=GridMaxX) and (NCLine.yStart=GridMaxY)) then     //reached the max vertex
						begin
							if (DebugFC) then PostObj.AddToNCFile('-----added a STEP at j='+inttostr(j)+#13#10+
																'GridMinX='+floattostr(GridMinX)+#13#10+
																'GridMinY='+floattostr(GridMinY)+#13#10+
																'GridMaxX='+floattostr(GridMaxX)+#13#10+
																'GridMaxY='+floattostr(GridMaxY)+#13#10);
							//step 2 happens when the path gets back to its original starting point
							//step 3 happens when the path reaches the farthest x and y position from the starting point, which have been recognized and stored before step 1 took place
							Inc(NumOfSteps);
							if (NumOfSteps=3) then begin
								if (DebugFC) then PostObj.AddToNCFile('----------second pass reached----------'+#13#10+
																	'flash cut starts at i='+inttostr(i)+#13#10+
																	'flash cut ends at j='+inttostr(j)+#13#10+
																	'GridMinX='+floattostr(GridMinX)+#13#10+
																	'GridMinY='+floattostr(GridMinY)+#13#10+
																	'GridMaxX='+floattostr(GridMaxX)+#13#10+
																	'GridMaxY='+floattostr(GridMaxY)+#13#10);
								ProcessFlashCut(i,j);
								IndexToSkip := j;
								break;
							end;
						end;
					end;
				end;
            end;
        end;
    end;
end;

var
	D1, d2, d3, d4, d5, d6	: double;
    dx, dy 					: double;
    Diagonal 				: boolean;
begin
	if (PostObj.DoingTime) then exit;	
    D1 := sqr((PostObj.x - PostObj.prex));
    D2 := sqr(PostObj.Y - PostObj.preY);
    D3 := sqrt (d1 + d2);
    Diagonal := True;
    DX := abs(PostObj.X - PostObj.Prex);
    DY := abs(PostObj.Y - PostObj.Prey);
    if  (DX < Diagonal_tolerance) or (DY < Diagonal_tolerance) then Diagonal := False;     
    PostObj.AddToNCFile('[PART_END]');

				PostObj.AddToNCFile('[PART_BEGIN]');  	// N per ogni icona

		end;

    end;

    //PostObj.AddToNCFile('N'+IntToStr(NCounter)+' G00,[X],[Y]');

procedure WriteLine;
var 
	PNCD 	: TNCData;
	PNCLine : TNCLine;
	i : integer;
	tmpString  : string;
	tmpString2 : string;
	KomRes : string;
    NextPNCD : TNCData;
begin
	FixCuttingTime;
	if (PostObj.DoingTime) then exit;	
    if (PostObj.ActiveNCPos+1<=PostObj.NCDataObjList.Count-1) then begin
		PNCD := PostObj.NCDataObjList.At_Object(PostObj.ActiveNCPos);
        NextPNCD := PostObj.NCDataObjList.At_Object(PostObj.ActiveNCPos+1);
        if (PNCD.EntSubAttr=27) {and (NextPNCD.Ignore)} and (FlashCutEnabled) and (FlashCutList.Count > 0) then begin
            for i:=0 to FlashCutList.Count-1 do begin
                KomRes := [' ', ' '];
                tmpString := FlashCutList[i];
                KomRes := LetterKomAndTrim(tmpString,';');
                tmpString := KomRes[0];
                tmpString2 := KomRes[1];
                if (PostObj.ActiveNCPos=StrToInt(tmpString2)) then begin
                    PostObj.AddToNCFile('[VAR8]');
                    KomRes := LetterKomAndTrim(tmpString,';');
                    tmpString := KomRes[0];
                    tmpString2 := KomRes[1];
                    PostObj.AddToNCFile(tmpString2);
                    KomRes := LetterKomAndTrim(tmpString,';');
                    tmpString := KomRes[0];
                    tmpString2 := KomRes[1];
                    PostObj.AddToNCFile(tmpString2);
                    KomRes := LetterKomAndTrim(tmpString,';');
                    tmpString := KomRes[0];
                    tmpString2 := KomRes[1];
                    //PostObj.AddToNCFile(tmpString2);
                    exit;
                end;
            end;
        end;
    end;
	PostObj.AddToNCFile('[VAR8]');
	if (PostObj.ToolNumber <> 10) then begin
		PostObj.AddToNCFile('[KERF]');
		PostObj.AddToNCFile('[FLAME]');
	end;
	PostObj.AddToNCFile('[OFFSET_OFF]');
	PostObj.AddToNCFile('[OFFSET_ON]');
	if (PostObj.ToolNumber = 10) then begin
		PostObj.AddToNCFile('[VAR8]');
		PostObj.AddToNCFile('[FLAME]');
	end else begin
		CheckNanoTab;	
	end;
	PNCD := PostObj.NCDataObjList.At_Object(PostObj.ActiveNCPos);
	if ((PNCD.ClassName = 'TNCLine') and (not PostObj.SinglePartProgram)) then begin
		PNCLine := PNCD;
		if (FormatFloat('0.00', PNCLine.xValAbs) = '0.00') and (FormatFloat('0.00', PNCLine.yValAbs) = '0.00') then begin
			ErrorList.Add('Attenzione! E'' stato programmato un G01 a coordinate (0;0).' + #13#10 + 'La generazione del file NC � stata annullata');
			PostObj.CriticalError := true;
		end;
	end;
	PostObj.AddToNCFile('[G],[X],[Y]');
	if (PostObj.ToolNumber <> 10) then CheckMinLength;
end;

procedure PunchPoint;
begin
	PostObj.AddToNCFile('E5');
	PostObj.AddToNCFile('M103');
	if (PunchPointType = 1) then begin
		PostObj.AddtoNCFile('G01,X' + FormatFloat('0.00', PostObj.X+CrossDim) + ',[Y]');
		PostObj.AddtoNCFile('G01,[X],Y' + FormatFloat('0.00', PostObj.Y+CrossDim));
		PostObj.AddtoNCFile('G01,[X],Y' + FormatFloat('0.00', PostObj.Y-CrossDim));
		PostObj.AddtoNCFile('G01,X' + FormatFloat('0.00', PostObj.X-CrossDim) + ',[Y]');
		PostObj.AddtoNCFile('G01,[X],[Y]');
	end;
end;

//************************************************************************************************************************
procedure WriteArc;
begin
	FixCuttingTime;
	if (PostObj.DoingTime) then exit;
	PostObj.AddToNCFile('[VAR8]');
	if (PostObj.ToolNumber <> 10) then begin
		PostObj.AddToNCFile('[KERF]');
		PostObj.AddToNCFile('[FLAME]');
	end;
	PostObj.AddToNCFile('[OFFSET_OFF]');
	PostObj.AddToNCFile('[OFFSET_ON]');
	if (PostObj.ToolNumber = 10) then begin
		PostObj.AddToNCFile('[VAR8]');
		PostObj.AddToNCFile('[FLAME]');
	end;
	if ((PostObj.ToolNumber <> 5) and (PostObj.ToolNumber <> 10)) then CheckNanoTab;
	if (PostObj.CircleSweepAngle > 1) then begin
		PostObj.AddToNCFile('[G],[X],[Y],[I],[J]');
	end else begin
		PostObj.AddToNCFile('G01,[X],[Y]');
	end;
	if (PostObj.ToolNumber <> 10) then CheckMinLength;
end;

procedure Amada_FO_End_Of_File;
begin
	if (PostObj.DoingTime) then exit;
	if (WritePercOnEndOfFile) then begin
		PostObj.AddToNCFile('%');
	end;
end;

procedure CreateSCHList;
var
	Listone 						: TStringList;
	StringList1 					: TStringList;
	s, NomeTask, NomePRG, Quante	: string;	
	Materiale, Spessore				: string;
	i								: integer;
begin
	if (PostObj.DoingTime) or (PostObj.CriticalError) then exit;
    //check conditions
    if (PostObj.SysOption5=1) and
       (PostObj.PostingMode=2) and
       ((PostObj.TotalNumOfLayOuts>1) or (AlwaysGenerateJobFile)) and
       (PostObj.CurrentLayOutNum=PostObj.TotalNumOfLayOuts) then
    begin
       //ok
    end else begin
       exit;
    end;
	
    StringList1 := TStringList.Create;
    Listone := TStringList.Create;
    StringList1.Clear;

    //write header
	Listone.Add('<schedule_info MachineName='+'"'+SCHMachineName+'">');
	//write body
    StringList1.LoadFromFile(VBFilePath);
    for i := 0 to StringList1.count-1 do begin
		s := stringlist1[i];
		kom(s,NomeTask);
		kom(s,NomePrg);
		kom(s,Quante);
		kom(s,Spessore);
		kom(s,Materiale);
		Listone.Add('  <parts_info>');
		Listone.Add('    <server_name>');
		Listone.Add('    </server_name>');
		Listone.Add('    <folder_name>');
		Listone.Add('    </folder_name>');
		Listone.Add('    <parts_name>'+NomePrg+'</parts_name>');
		Listone.Add('    <product_quantity>'+Quante+'</product_quantity>');
		Listone.Add('    <reception_quantity>0</reception_quantity>');
		Listone.Add('    <rejection_quantity>0</rejection_quantity>');
		Listone.Add('    <complete_code>2</complete_code>');
		Listone.Add('    <thickness>'+Spessore+'</thickness>');
		Listone.Add('    <material_name>'+Materiale+'</material_name>');
		Listone.Add('    <punch_number>0</punch_number>');
		Listone.Add('    <die_number>0</die_number>');
		Listone.Add('    <utilization>100</utilization>');
		Listone.Add('  </parts_info>');
	end;
	Listone.Add('</schedule_info>');
	DeleteFile(VBFilePath);
	Listone.SaveToFile(IncludeTrailingBackSlash(IncludeTrailingBackSlash(PostObj.NCPath))+PostObj.TaskName+'.SCH');
	Listone.Free;
    StringList1.Free;
end;
